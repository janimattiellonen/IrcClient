  I recommend adding three methods to the ChannelManager class for user
  management:

  1. addUserToChannel(channelName: string, user: User): void

  - Adds a user to a channel's user list
  - Should check if the channel exists
  - Should check if the user already exists in the channel (to avoid
  duplicates)

  2. removeUserFromChannel(channelName: string, userNick: string): void

  - Removes a user from a channel's user list
  - Should check if the channel exists
  - Uses userNick as the identifier since nick is the unique identifier in
  IRC

  3. updateUserInChannel(channelName: string, oldNick: string, updatedUser:
  User): void

  - Updates a user's information in a channel (useful for nick changes, mode
   changes, etc.)
  - Should check if the channel exists
  - Should check if the old user exists

  Implementation would look like:

  addUserToChannel(channelName: string, user: User): void {
    const channel = this.channels[channelName];
    if (!channel) {
      throw new Error(`Channel ${channelName} does not exist`);
    }

    const userExists = channel.users.some(u => u.nick === user.nick);
    if (userExists) {
      throw new Error(`User ${user.nick} already exists in channel
  ${channelName}`);
    }

    channel.users.push(user);
  }

  removeUserFromChannel(channelName: string, userNick: string): void {
    const channel = this.channels[channelName];
    if (!channel) {
      throw new Error(`Channel ${channelName} does not exist`);
    }

    const userIndex = channel.users.findIndex(u => u.nick === userNick);
    if (userIndex === -1) {
      throw new Error(`User ${userNick} not found in channel
  ${channelName}`);
    }

    channel.users.splice(userIndex, 1);
  }

  updateUserInChannel(channelName: string, oldNick: string, updatedUser:
  User): void {
    const channel = this.channels[channelName];
    if (!channel) {
      throw new Error(`Channel ${channelName} does not exist`);
    }

    const userIndex = channel.users.findIndex(u => u.nick === oldNick);
    if (userIndex === -1) {
      throw new Error(`User ${oldNick} not found in channel
  ${channelName}`);
    }

    channel.users[userIndex] = updatedUser;
  }
