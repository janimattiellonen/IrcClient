  server/src/irc/

  Layer 1: Transport (tcpClient.ts)

  - Map web clients to IRC connections

  - Route incoming messages to appropriate handlers
  - Delegate to connection manager
  - Stay "dumb" - no protocol details

  Layer 1: Transport (transport/tcpClient.ts)

  import * as net from 'net';

  type TcpConnectionParams = {
    host: string;
    port: number;
  };

  type TcpClientEvents = {
    onConnect: () => void;
    onData: (data: Buffer) => void;
    onError: (error: Error) => void;
    onClose: () => void;
  };

  export class TcpClient {
    private socket: net.Socket | null = null;

    connect(params: TcpConnectionParams, events: TcpClientEvents): void {
      this.socket = net.createConnection({
        host: params.host,
        port: params.port
      });

      this.socket.on('connect', events.onConnect);
      this.socket.on('data', events.onData);
      this.socket.on('error', events.onError);
      this.socket.on('close', events.onClose);
    }

    send(data: string): void {
      if (!this.socket) {
        throw new Error('Socket not connected');
      }
      this.socket.write(data);
    }

    disconnect(): void {
      this.socket?.end();
      this.socket = null;
    }

    isConnected(): boolean {
      return this.socket !== null && !this.socket.destroyed;
    }
  }

  // IRC message formatting utilities
  export const IrcProtocol = {
    // Format IRC commands (NO socket interaction here)
    formatNick: (nickname: string): string => {
      return `NICK ${nickname}\r\n`;
    },

    formatUser: (username: string, realname: string): string => {
      return `USER ${username} 0 * :${realname}\r\n`;
    },

    formatJoin: (channel: string): string => {
      return `JOIN ${channel}\r\n`;
    },

    formatPrivMsg: (target: string, message: string): string => {
      return `PRIVMSG ${target} :${message}\r\n`;
    },

    formatPong: (server: string): string => {
      return `PONG :${server}\r\n`;
    },

    // Parse incoming IRC messages
    parseMessage: (raw: string) => {
      // Basic IRC message parsing
      // Returns structured data like { prefix, command, params }
      const match = raw.match(/^(?::(\S+) )?(\S+)(?: (.+))?$/);
      if (!match) return null;

      return {
        prefix: match[1] || null,
        command: match[2],
        params: match[3] ? match[3].split(' ') : []
      };
    }
  };

  Layer 2: Protocol (protocol/ircConnection.ts)

  import { TcpClient } from '../transport/tcpClient';
  import { IrcProtocol } from './ircProtocol';

  type IrcConnectionConfig = {
    host: string;
    port: number;
    nickname: string;
    username?: string;
    realname?: string;
  };

  type IrcConnectionEvents = {
    onRegistered: () => void;
    onMessage: (data: any) => void;
    onError: (error: Error) => void;
    onDisconnect: () => void;
  };

  export class IrcConnection {
    private tcpClient: TcpClient;
    private config: IrcConnectionConfig;
    private registered = false;

    constructor(config: IrcConnectionConfig) {
      this.config = config;
      this.tcpClient = new TcpClient();
    }

    connect(events: IrcConnectionEvents): void {
      this.tcpClient.connect(
        { host: this.config.host, port: this.config.port },
        {
          onConnect: () => {
            // Send IRC registration commands
            this.sendNick(this.config.nickname);
            this.sendUser(
              this.config.username || this.config.nickname,
              this.config.realname || 'IRC User'
            );
          },
          onData: (buffer: Buffer) => {
            const lines = buffer.toString().split('\r\n');
            for (const line of lines) {
              if (!line) continue;
              this.handleIrcMessage(line, events);
            }
          },
          onError: events.onError,
          onClose: events.onDisconnect
        }
      );
    }

    private handleIrcMessage(raw: string, events: IrcConnectionEvents): void {
      const parsed = IrcProtocol.parseMessage(raw);
      if (!parsed) return;

      // Handle PING automatically
      if (parsed.command === 'PING') {
        const server = parsed.params[0] || '';
        this.tcpClient.send(IrcProtocol.formatPong(server));
        return;
      }

      // Check for successful registration (RPL_WELCOME = 001)
      if (parsed.command === '001' && !this.registered) {
        this.registered = true;
        events.onRegistered();
      }

      // Forward all messages to application layer
      events.onMessage({ raw, parsed });
    }

    // IRC command methods (use protocol formatting)
    sendNick(nickname: string): void {
      this.tcpClient.send(IrcProtocol.formatNick(nickname));
    }

    sendUser(username: string, realname: string): void {
      this.tcpClient.send(IrcProtocol.formatUser(username, realname));
    }

    joinChannel(channel: string): void {
      if (!this.registered) {
        throw new Error('Cannot join channel before registration complete');
      }
      this.tcpClient.send(IrcProtocol.formatJoin(channel));
    }

    sendMessage(target: string, message: string): void {
      if (!this.registered) {
        throw new Error('Cannot send message before registration complete');
      }
      this.tcpClient.send(IrcProtocol.formatPrivMsg(target, message));
    }

    disconnect(): void {
      this.tcpClient.disconnect();
      this.registered = false;
    }

    isConnected(): boolean {
      return this.tcpClient.isConnected();
    }
  }

  type ConnectionInfo = {
    connection: IrcConnection;
    nickname: string;
    server: string;
    registered: boolean;

  export class IrcConnectionManager {
    // Map Socket.IO client ID to IRC connection
    private connections = new Map<string, ConnectionInfo>();

    createConnection(
      clientId: string,
      config: { host: string; port: number; nickname: string }
    ): void {
      // Check if connection already exists
      if (this.connections.has(clientId)) {

      const ircConnection = new IrcConnection({

      const connectionInfo: ConnectionInfo = {

      if (info) {

      for (const [clientId] of this.connections) {
        this.disconnectClient(clientId);

  import { AppMessage, MESSAGE_JOIN_CHANNEL, MESSAGE_LOGIN } from 'shared/messageTypes';
  import { IrcConnectionManager } from './ircConnectionManager';

  // This would be passed from server.ts
  let connectionManager: IrcConnectionManager;

  export function initializeMessageHandler(manager: IrcConnectionManager): void {
    connectionManager = manager;
  }

  export function handleClientMessage(
    message: AppMessage,
    clientId: string
  ): void {
    // Clean routing - no protocol details here
    if (message.type === MESSAGE_LOGIN) {
      handleLogin(message.payload, clientId);
    } else if (message.type === MESSAGE_JOIN_CHANNEL) {
      handleJoinChannel(message.payload, clientId);
    }
  }

  function handleLogin(
    payload: { server: string; port: number; nickname: string },
    clientId: string
  ): void {
    // Delegate to connection manager - no IRC protocol here
    connectionManager.createConnection(clientId, {
      host: payload.server,
      port: payload.port,
      nickname: payload.nickname
    });
  }

  function handleJoinChannel(
    payload: { channel: string },
    clientId: string
  ): void {
    const connection = connectionManager.getConnection(clientId);
    if (!connection) {
      throw new Error('No IRC connection found for client');
    }

    // Delegate to IRC connection - no protocol formatting here
    connection.joinChannel(payload.channel);
  }
